JAVA_TOOL_OPTIONS="-Ddrmd.test.u07ProofDir=$proof_dir" ./gradlew --offline --no-daemon \
  -I scripts/v8/focused_gametest.init.gradle \
  "-PdrmdGameTests=$(cat docs/v8/EVIDENCE/task20-20261119/union-selector.txt)" \
  test --tests '*VolumetricBoundsTest' --tests '*D6ChunkPosTest' \
  --tests '*PlanetaryZonesTest' --tests '*GravityNoticeGateTest' \
  --tests '*GravityFieldIdsTest' --tests '*BoundedSpatialIndexTest' \
